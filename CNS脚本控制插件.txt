    function parseResult(res) {
        if (res === null || res === undefined) return '(无输出)';
        if (typeof res === 'object' && res.content !== undefined) return String(res.content).trim();
        if (typeof res === 'string') return res.trim();
        try { return JSON.stringify(res); } catch(e) { return String(res); }
    }

    function nowTimeStr() {
        const now = new Date();
        return `${now.getFullYear()}/${String(now.getMonth()+1).padStart(2,'0')}/${String(now.getDate()).padStart(2,'0')} ${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}:${String(now.getSeconds()).padStart(2,'0')}`;
    }

    }

    // shell 单引号安全转义，防止链接/文件名里的特殊字符被当成命令
    function shQuote(s) { return "'" + String(s).replace(/'/g, "'\\''") + "'"; }

    function isValidHttpUrl(u) { return /^https?:\/\/[^\s'"\\<>]+$/i.test(u); }

    function fmtSize(bytes) {
        const n = Number(bytes) || 0;
        if (n < 1024) return n + ' B';
        if (n < 1048576) return (n / 1024).toFixed(1) + ' KB';
        return (n / 1048576).toFixed(2) + ' MB';
    }

    // 读文本文件：项目约定不用 cat —— timeout 2s awk 兜底，避免大文件/特殊文件把 shell 卡住
    async function readFileSafe(path, timeout) {
        const res = await runShellWithRoot(`timeout 2s awk '{print}' ${shQuote(path)} 2>/dev/null`, timeout || 15000);
        return parseResult(res);
    }

    // 分块写入 base64 字符串（单参数 >128KB 会 E2BIG，必须分块）
    async function writeB64String(targetPath, b64) {
        const tmp = targetPath + '.b64';
        await runShellWithRoot(`rm -f ${shQuote(tmp)}`);
        await runShellWithRoot(`: > ${shQuote(tmp)}`);
        for (let i = 0; i < b64.length; i += B64_CHUNK) {
            const part = b64.slice(i, i + B64_CHUNK);
            const res = await runShellWithRoot(`printf '%s' '${part}' >> ${shQuote(tmp)}`);
            const out = parseResult(res);
            if (out && /no such|not found|permission|denied|error|失败/i.test(out)) {
                await runShellWithRoot(`rm -f ${shQuote(tmp)}`);
                return { ok: false, out };
            }
        }
        const res = await runShellWithRoot(
            `(base64 -d ${shQuote(tmp)} > ${shQuote(targetPath)} 2>/dev/null || busybox base64 -d ${shQuote(tmp)} > ${shQuote(targetPath)}) && rm -f ${shQuote(tmp)} && echo __OK__`,
            60000
        );
        const out = parseResult(res);
        return { ok: out.includes('__OK__'), out: out.replace('__OK__', '').trim() };
    }

    async function writeTextFile(targetPath, text) {
        return writeB64String(targetPath, utf8ToBase64(text));
    }

    // 探测可用的 curl（设备上 UFI-TOOLS 自带的是绝对路径，PATH 里未必有）

    async function findCurlBin() {
        if (_curlBin !== null) return _curlBin;
        const res = await runShellWithRoot(
            `for c in /data/data/com.minikano.f50_sms/files/curl /system/bin/curl /data/adb/ap/bin/curl; do [ -x "$c" ] && echo "$c" && exit 0; done; command -v curl 2>/dev/null || echo ''`,

        _curlBin = parseResult(res).split('\n').map(s => s.trim()).filter(Boolean)[0] || '';
        return _curlBin;
    }

    // 杀 CNS 进程：按进程名精确杀 + /proc/exe 兜底，明确跳过自身与父进程，
    // 不能用 pkill -f '<CNS目录>'（那会把执行这条命令的 shell 自己也匹配掉）
    async function killCnsProcs() {
        const res = await runShellWithRoot(
            `pkill -x ${shQuote(CNS_PROCESS_NAME)} 2>/dev/null; ` +
            `for p in /proc/[0-9]*; do pid=\${p##*/}; ` +
            `[ "$pid" = "$$" ] && continue; [ "$pid" = "$PPID" ] && continue; ` +
            `case "$(readlink "$p/exe" 2>/dev/null)" in */${CNS_PROCESS_NAME}) kill "$pid" 2>/dev/null;; esac; ` +
            `done; echo __DONE__`,
            25000
        );
        return parseResult(res).includes('__DONE__');
    }

    // 多步骤日志：add 追加一行，progress 原地刷新最后一行
    function makeStepLogger() {
        const lines = [];
        let progressIdx = -1;
        const render = () => appendLog(lines.join('\n'));
        return {
            add(text) { progressIdx = -1; lines.push(text); render(); },
            progress(text) {
                if (progressIdx < 0) { lines.push(text); progressIdx = lines.length - 1; }
                else lines[progressIdx] = text;
                render();
            }
        };
    }

    async function isCnsRunning() {
        const scriptChk = await runShellWithRoot(`[ -f ${shQuote(CNS_DIR + '/start.sh')} ] && echo EXISTS || echo NO`);
        const noScript = !parseResult(scriptChk).includes('EXISTS');

        let res = await runShellWithRoot(`pgrep -x ${shQuote(CNS_PROCESS_NAME)} 2>/dev/null`);
        let out = parseResult(res).trim();
        if (/\d/.test(out)) {
            return { running: true, info: 'PID: ' + out.split(/\s+/).join(', '), noScript };
        }

        res = await runShellWithRoot(`pgrep -f ${shQuote(CNS_DIR)} 2>/dev/null`);
        out = parseResult(res).trim();
        if (/\d/.test(out)) {
            return { running: true, info: 'PID: ' + out.split(/\s+/).join(', '), noScript };
        }

        res = await runShellWithRoot(`ps 2>/dev/null | grep ${shQuote(CNS_DIR)} | grep -v grep`);
        out = parseResult(res).trim();
        if (out) {
            const pidMatch = out.match(/^\s*\S+\s+(\d+)/);
            const pid = pidMatch ? pidMatch[1] : '?';
            return { running: true, info: 'PID: ' + pid, noScript };
        }

        return { running: false, info: '', noScript };
    }

    function statusText(r) {
        if (r.noScript) return '未检测到 CNS 脚本，点「下载 CNS」可一键安装';
        return r.running ? 'CNS 正在运行' + (r.info ? ' (' + r.info + ')' : '') : 'CNS 未运行';
    }

    async function runCnsScript(scriptName, actionDesc) {
        let path;
        if (scriptName === '生成配置.sh') {
            path = CNS_UP_SCRIPT;
        } else {
            path = CNS_DIR + '/' + scriptName;
        }
        const check = await runShellWithRoot(`[ -f ${shQuote(path)} ] && echo EXISTS || echo NO`);
        if (!parseResult(check).includes('EXISTS')) {
            appendLog(`脚本 ${path} 不存在，无法${actionDesc || '执行'}`);
            return;
        }
        appendLog(`正在${actionDesc || '执行'} ${path} ...`);
        const res = await runShellWithRoot(`sh ${shQuote(path)}`, 30000);
        appendLog(parseResult(res));
        // 不再根据脚本输出里的关键字猜运行状态（"not running/未运行" 会被 /run/ 误判成运行中），
        // 统一以进程探测为准
        setTimeout(async () => {
            const st = await isCnsRunning();
            updateStatus(st.running);
        }, 1500);
    }

    // --- 检测公网IP ---
    async function checkPublicIP() {
        const tip = '正在检测公网IP (ip.3322.net) ...';
        appendLog(tip);
        const curlBin = await findCurlBin();
        let ip = '';
        if (curlBin) {
            const res = await runShellWithRoot(`${shQuote(curlBin)} -s --connect-timeout 10 ip.3322.net 2>/dev/null`, 15000);
            ip = parseResult(res).trim();
        }
        if (!ip || /error|fail|not found|timeout/i.test(ip)) {
            const res = await runShellWithRoot(`wget -q -O - --timeout=10 ip.3322.net 2>/dev/null`, 15000);
            ip = parseResult(res).trim();
        }
        if (ip && !/error|fail|not found|timeout/i.test(ip) && /^\d+\.\d+\.\d+\.\d+$/.test(ip)) {
            appendLog('公网IP: ' + ip);
        } else {
            appendLog('获取IP失败，可能无网络连接或服务不可用' + (curlBin ? '' : '（未找到 curl）'));
        }
    }

        const fail = (msg, toast) => {
            log.add(msg + '\n(临时文件保留在 ' + DL_TMP_DIR + '，可重试或手动清理)');
            showConfigToast(toast || '下载失败');
        };
        try {
            log.add('下载地址: ' + url);
            await runShellWithRoot(`rm -rf ${shQuote(DL_TMP_DIR)}; mkdir -p ${shQuote(DL_TMP_DIR)}`);

            const curlBin = await findCurlBin();
            if (curlBin) log.add('使用下载工具: ' + curlBin);

            // 后台下载脚本：避免长时间下载被前端超时打断
            const dlScript = [
                '#!/bin/sh',
                'URL=' + shQuote(url),
                'OUT=' + shQuote(DL_ZIP),
                'LOG=' + shQuote(DL_LOG),
                'CODE=' + shQuote(DL_CODE),
                'CURL=' + shQuote(curlBin || ''),
                'if [ -n "$CURL" ]; then',
                '  "$CURL" -fsS -L --connect-timeout 15 -o "$OUT" "$URL" 2>"$LOG"',
                '  echo $? > "$CODE"',
                'elif command -v wget >/dev/null 2>&1; then',
                '  wget -T 20 -O "$OUT" "$URL" 2>"$LOG"',
                '  echo $? > "$CODE"',
                'else',
                '  echo "curl / wget 均不可用" > "$LOG"',
                '  echo 127 > "$CODE"',
                'fi',
                ''
            ].join('\n');
            await writeTextFile(DL_SCRIPT, dlScript);

            const started = parseResult(await runShellWithRoot(
                `if command -v nohup >/dev/null 2>&1; then (nohup sh ${shQuote(DL_SCRIPT)} >/dev/null 2>&1 &); else (sh ${shQuote(DL_SCRIPT)} >/dev/null 2>&1 &); fi; echo STARTED`
            ));
            if (!started.includes('STARTED')) return fail('× 无法启动下载任务');

            let waited = 0, code = null;
            while (waited < DL_MAX_WAIT_MS) {
                await sleep(2000);
                waited += 2000;
                const st = parseResult(await runShellWithRoot(
                    `if [ -f ${shQuote(DL_CODE)} ]; then printf 'DONE:'; timeout 2s awk '{print}' ${shQuote(DL_CODE)}; elif [ -f ${shQuote(DL_ZIP)} ]; then printf 'SIZE:'; wc -c < ${shQuote(DL_ZIP)}; else printf 'SIZE:0'; fi`
                )).trim();
                if (st.indexOf('DONE:') === 0) {
                    code = parseInt(st.replace(/[^0-9-]/g, ''), 10);
                    break;
                }
                const size = parseInt(st.replace(/[^0-9]/g, ''), 10) || 0;
                log.progress('下载中... 已接收 ' + fmtSize(size) + '  (' + Math.round(waited / 1000) + 's)');
            }

            if (code === null) {
                await runShellWithRoot(`pkill -f ${shQuote(DL_SCRIPT)} 2>/dev/null; pkill -f ${shQuote(DL_ZIP)} 2>/dev/null`);
                return fail('× 下载超时（超过 ' + Math.round(DL_MAX_WAIT_MS / 1000) + ' 秒），已中止', '下载超时');
            }
            if (code !== 0) {
                const errLog = parseResult(await runShellWithRoot(`head -c 500 ${shQuote(DL_LOG)} 2>/dev/null`));
                return fail('× 下载失败 (退出码 ' + code + ')' + (errLog && errLog !== '(无输出)' ? '\n' + errLog : ''));
            }

            const sizeOut = parseResult(await runShellWithRoot(
                `if [ -f ${shQuote(DL_ZIP)} ]; then wc -c < ${shQuote(DL_ZIP)}; else echo 0; fi`
            )).replace(/[^0-9]/g, '');
            const zipSize = parseInt(sizeOut, 10) || 0;
            if (zipSize <= 0) return fail('× 下载文件为空，请检查链接');
            log.add('✓ 下载完成，已重命名为 cns.zip (' + fmtSize(zipSize) + ')');

            // 校验是否为 zip（PK 头），避免把网页/重定向页当成压缩包
            const magic = parseResult(await runShellWithRoot(`head -c 2 ${shQuote(DL_ZIP)} 2>/dev/null`));
            if (magic.indexOf('PK') !== 0) {
                const preview = parseResult(await runShellWithRoot(`head -c 200 ${shQuote(DL_ZIP)} 2>/dev/null`));
                return fail('× 下载内容不是 zip 压缩包，请确认是直链下载地址\n内容预览: ' + preview, '不是 zip 文件');
            }

            log.add('正在解压...');
            const tool = parseResult(await runShellWithRoot(
                `if command -v unzip >/dev/null 2>&1; then echo UNZIP; elif command -v busybox >/dev/null 2>&1; then echo BUSYBOX; else echo NONE; fi`
            ));
            if (tool.includes('NONE')) return fail('× 设备上没有 unzip（也没有 busybox），无法解压', '缺少 unzip');
            const unzipBin = tool.includes('UNZIP') ? 'unzip' : 'busybox unzip';
            await runShellWithRoot(`mkdir -p ${shQuote(DL_EXTRACT)}`);
            const unzipOut = parseResult(await runShellWithRoot(
                `${unzipBin} -o ${shQuote(DL_ZIP)} -d ${shQuote(DL_EXTRACT)} 2>&1`, 120000
            ));
            const entries = parseResult(await runShellWithRoot(`ls -A ${shQuote(DL_EXTRACT)} 2>/dev/null`))
                .split('\n').map(s => s.trim())
                .filter(s => s && s !== '__MACOSX' && s !== '.DS_Store' && s !== '(无输出)');
            if (!entries.length) return fail('× 解压失败\n' + unzipOut, '解压失败');

            // 压缩包内只有一个顶层目录时，取该目录作为 cns 目录内容
            let payload = DL_EXTRACT;
            if (entries.length === 1) {
                const only = DL_EXTRACT + '/' + entries[0];
                const isDir = parseResult(await runShellWithRoot(`[ -d ${shQuote(only)} ] && echo DIR || echo FILE`));
                if (isDir.includes('DIR')) payload = only;
            }
            log.add('✓ 解压完成' + (payload === DL_EXTRACT ? '' : '（顶层目录: ' + entries[0] + '）'));

            const dirExists = parseResult(await runShellWithRoot(`[ -d ${shQuote(CNS_DIR)} ] && echo EXISTS || echo NO`)).includes('EXISTS');
            if (dirExists) {
                if (!(await showCnsConfirm(CNS_DIR + ' 已存在。\n继续将停止运行中的 CNS 并删除原目录后覆盖安装（原配置会丢失），确定吗？', '覆盖安装', true))) {
                    log.add('已取消安装，下载文件保留在 ' + DL_ZIP);
                    showConfigToast('已取消安装');
                    return;
                }
                log.add('正在停止旧的 CNS 并删除原目录...');
                await runShellWithRoot(`if [ -f ${shQuote(CNS_DIR + '/stop.sh')} ]; then sh ${shQuote(CNS_DIR + '/stop.sh')}; fi`);
                await killCnsProcs();
                await runShellWithRoot(`rm -rf ${shQuote(CNS_DIR)}`);
            }

            const mvOut = parseResult(await runShellWithRoot(`mv ${shQuote(payload)} ${shQuote(CNS_DIR)} 2>&1`));
            const installed = parseResult(await runShellWithRoot(`[ -d ${shQuote(CNS_DIR)} ] && echo EXISTS || echo NO`)).includes('EXISTS');
            if (!installed) return fail('× 移动到 ' + CNS_DIR + ' 失败' + (mvOut && mvOut !== '(无输出)' ? '\n' + mvOut : ''), '安装失败');
            log.add('✓ 文件夹已放入 /data/ 并命名为 cns');

            const chmodOut = parseResult(await runShellWithRoot(`chmod -R 777 ${shQuote(CNS_DIR)} 2>&1`));
            if (chmodOut && chmodOut !== '(无输出)' && /denied|permission|not permitted|error|失败/i.test(chmodOut)) {
                log.add('△ 赋权提示: ' + chmodOut);
            } else {
                log.add('✓ 已对 ' + CNS_DIR + ' 内所有文件赋予 777 权限');
            }

            const fileList = parseResult(await runShellWithRoot(`ls -A ${shQuote(CNS_DIR)} 2>/dev/null | head -n 40`));
            log.add('目录内容:\n' + fileList);

            await runShellWithRoot(`rm -rf ${shQuote(DL_TMP_DIR)}`);
            try { localStorage.setItem(DL_URL_KEY, url); } catch(e) {}
            showConfigToast('CNS 安装完成');

            const infoEl = document.getElementById('cns_info');
            if (infoEl) infoEl.textContent = ' 已安装';
            const st = await isCnsRunning();
            updateStatus(st.running);
            updateAutostartBtn(await isAutostartEnabled());
        } catch(e) {
            fail('× 下载安装出错: ' + String(e));
        } finally {
            if (btn) btn.disabled = false;
        }
    }

    async function isAutostartEnabled() {
        // 用子串匹配，新旧两种自启行写法都能识别
        const res = await runShellWithRoot(`grep -qF ${shQuote(AUTOSTART_KEY)} ${shQuote(AUTOSTART_SCRIPT)} && echo EXISTS || echo NO`);
        return parseResult(res).includes('EXISTS');
    }

    async function setAutostart(enable) {
        if (enable) {
            // 先清掉所有引用 CNS 自启的历史行（旧格式带引号/条件判断的也算），再写入规范行 → 幂等
            await runShellWithRoot(
                `[ -f ${shQuote(AUTOSTART_SCRIPT)} ] && grep -vF ${shQuote(AUTOSTART_KEY)} ${shQuote(AUTOSTART_SCRIPT)} > ${shQuote(AUTOSTART_SCRIPT + '.tmp')}; ` +
                `[ -f ${shQuote(AUTOSTART_SCRIPT + '.tmp')} ] && mv -f ${shQuote(AUTOSTART_SCRIPT + '.tmp')} ${shQuote(AUTOSTART_SCRIPT)}; ` +
                `grep -qxF ${shQuote(AUTOSTART_LINE)} ${shQuote(AUTOSTART_SCRIPT)} 2>/dev/null || echo ${shQuote(AUTOSTART_LINE)} >> ${shQuote(AUTOSTART_SCRIPT)}`,
                20000
            );
        } else {
            // 不依赖 sed（\\|regex|d 在 toybox/BSD sed 上不保证生效）：grep -v 过滤后覆盖回去
            await runShellWithRoot(
                `[ -f ${shQuote(AUTOSTART_SCRIPT)} ] && grep -vF ${shQuote(AUTOSTART_KEY)} ${shQuote(AUTOSTART_SCRIPT)} > ${shQuote(AUTOSTART_SCRIPT + '.tmp')}; ` +
                `[ -f ${shQuote(AUTOSTART_SCRIPT + '.tmp')} ] && mv -f ${shQuote(AUTOSTART_SCRIPT + '.tmp')} ${shQuote(AUTOSTART_SCRIPT)}`,
                20000
            );
        }
        const on = await isAutostartEnabled();
        updateAutostartBtn(on);
        appendLog(enable
            ? '开机自启已开启（写入 ' + AUTOSTART_SCRIPT + '，开机后延迟 ' + AUTOSTART_DELAY_SEC + ' 秒启动）\n' + AUTOSTART_LINE
            : '开机自启已关闭');
        showConfigToast(enable ? '自启已开启' : '自启已关闭');
    }

    function utf8ToBase64(str) {
        if (window.TextEncoder) {
            const bytes = new TextEncoder().encode(str);
            let bin = '';
            for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
            return btoa(bin);
        }
        return btoa(unescape(encodeURIComponent(str)));
    }
    function arrayBufferToBase64(buffer) {
        const bytes = new Uint8Array(buffer);
        let bin = '';
        for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
        return btoa(bin);
    }
    function readFileAsBase64(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.onload = () => {
                try { resolve(arrayBufferToBase64(reader.result)); } catch(e) { reject(e); }
            };
            reader.onerror = () => reject(reader.error || new Error('读取文件失败'));
            reader.readAsArrayBuffer(file);
        });
    }

    // 下拉选项一律用 DOM 拼，避免文件名里的 <>&" 把列表拼坏
    function setSelectOptions(sel, files) {
        if (!sel) return;
        sel.textContent = '';
        if (!files.length) {
            const opt = document.createElement('option');
            opt.value = '';
            opt.textContent = '无配置文件';
            sel.appendChild(opt);

            sel.appendChild(opt);
        }
    }

    // 列出 <CNS目录> 下的全部文件（递归），name 用相对路径，选中即可直接编辑
    async function fillConfigSelect() {
        const sel = document.getElementById('cns_config_select');
        const files = [];
        const res = await runShellWithRoot(
            `cd ${shQuote(CNS_DIR)} && find . -type f 2>/dev/null | head -n ${FILE_LIST_MAX} || echo ''`
        );
        const rels = parseResult(res).split('\n')
            .map(s => s.trim().replace(/^\.\//, ''))
            .filter(s => s && s !== '(无输出)' && s !== '无输出');
        for (const rel of rels) {
            if (files.some(item => item.name === rel)) continue;
            files.push({ name: rel, path: CNS_DIR + '/' + rel });
        }
        files.sort((a, b) => {
            if (a.name === 'config.ini') return -1;
            if (b.name === 'config.ini') return 1;
            if (a.name === 'makeCNSMode/cns配置.ini') return -1;
            if (b.name === 'makeCNSMode/cns配置.ini') return 1;
            return a.name.localeCompare(b.name);
        });
        setSelectOptions(sel, files);
        return files;
    }

    // 下拉里的 name 是 <CNS目录> 下的相对路径，统一拼成绝对路径
    function configPath(rel) { return CNS_DIR + '/' + String(rel).replace(/^\.\//, ''); }

    // 读取前体检：不存在 / 超过上限 / 含 NUL（二进制）都不加载，免得把 WebView 卡死
    async function probeConfigFile(path) {
        const res = await runShellWithRoot(
            `p=${shQuote(path)}; ` +
            `if [ ! -f "$p" ]; then echo MISSING; ` +
            `elif [ $(wc -c < "$p" 2>/dev/null || echo 0) -gt ${FILE_MAX_EDIT_BYTES} ]; then echo TOO_BIG; ` +
            `elif od -An -c -N 256 "$p" 2>/dev/null | grep -qF '\\0'; then echo BINARY; ` +
            `else echo TEXT; fi`
        );
        const out = parseResult(res);
        if (out.includes('MISSING')) return 'missing';
        if (out.includes('TOO_BIG')) return 'too_big';
        if (out.includes('BINARY')) return 'binary';
        return 'text';
    }

    async function loadConfigFile(fileName) {
        const ta = document.getElementById('cns_config_text');
        if (!ta || !fileName) return;
        configCurrentFile = fileName;
        updateConfigBadge(fileName);
        ta.disabled = true;
        ta.value = '';
        configSavedContent = '';
        setConfigFeedback('正在读取 ' + fileName + ' ...', true);
        const path = configPath(fileName);
        try {
            const kind = await probeConfigFile(path);
            if (kind === 'missing') { setConfigFeedback('文件不存在: ' + fileName, false); return; }
            if (kind === 'too_big') { setConfigFeedback(fileName + ' 超过 ' + fmtSize(FILE_MAX_EDIT_BYTES) + '，不在编辑器里加载', false); return; }
            if (kind === 'binary') { setConfigFeedback(fileName + ' 是二进制文件，不在编辑器里加载', false); return; }
            const text = await readFileSafe(path);
            ta.value = text;
            // 基准取「textarea 里的值」，不再拿 trim 过的文件原文当基准 ——
            // 否则文件结尾多一个换行就会误报「有未保存的修改」
            configSavedContent = ta.value;
            setConfigFeedback(text === '' ? (fileName + ' 是空文件') : ('已加载 ' + fileName), true);
        } catch(e) {
            configSavedContent = '';
            setConfigFeedback('读取失败: ' + String(e), false);
        } finally {
            ta.disabled = false;
        }
    }

    async function saveConfigFile() {
        const ta = document.getElementById('cns_config_text');
        const fileName = configCurrentFile;
        if (!ta || !fileName) return;
        const btn = document.getElementById('btn_config_save');
        if (btn) btn.disabled = true;
        setConfigFeedback('正在保存 ' + fileName + ' ...', true);
        const path = configPath(fileName);
        // 相对路径可能带子目录，保存前先把父目录建出来
        await runShellWithRoot(`mkdir -p ${shQuote(path.replace(/\/[^/]*$/, ''))}`);
        try {
            const ret = await writeTextFile(path, ta.value);
            if (!ret.ok) {
                setConfigFeedback('保存失败: ' + (ret.out || '写入未完成'), false);
                appendLog('保存配置失败[' + fileName + ']: ' + (ret.out || '写入未完成'));
                return;
            }
            configSavedContent = ta.value;
            setConfigFeedback(fileName + ' 保存成功', true);
            appendLog('配置 ' + fileName + ' 已保存');
            showConfigToast('修改成功');
            const modal = document.getElementById('cns_config_modal');
            if (modal) modal.remove();
        } catch(e) {
            setConfigFeedback('保存失败: ' + String(e), false);
            appendLog('保存配置出错[' + fileName + ']: ' + String(e));
        } finally {
            if (btn) btn.disabled = false;
        }
    }

    async function deleteConfigFile() {
        const fileName = configCurrentFile;
        if (!fileName) return;
        if (!(await showCnsConfirm('确定删除 ' + fileName + ' 吗？\n删除后不可恢复！', '删 除', true))) return;
        const btn = document.getElementById('btn_config_delete');
        if (btn) btn.disabled = true;
        setConfigFeedback('正在删除 ' + fileName + ' ...', true);
        const path = configPath(fileName);
        try {
            const res = await runShellWithRoot(`rm -f ${shQuote(path)}`);
            const out = parseResult(res).trim();
            if (out && /not found|no such|permission|denied|error|失败/i.test(out)) {
                setConfigFeedback('删除失败: ' + out, false);
                appendLog('删除配置失败[' + fileName + ']: ' + out);
                return;
            }
            appendLog('配置 ' + fileName + ' 已删除');
            showConfigToast('删除成功');
            const modal = document.getElementById('cns_config_modal');
            if (modal) modal.remove();
        } catch(e) {
            setConfigFeedback('删除失败: ' + String(e), false);
            appendLog('删除配置出错[' + fileName + ']: ' + String(e));
        } finally {
            if (btn) btn.disabled = false;
        }
    }

    async function closeConfigModal() {
        const modal = document.getElementById('cns_config_modal');
        if (!modal) return;

    function pickConfigFiles() {
        const input = document.createElement('input');
        input.type = 'file';

        document.body.appendChild(input);

    async function uploadConfigFiles(picked) {
        if (!picked.length) return;
        appendLog('开始上传 ' + picked.length + ' 个文件...');
        const results = [];
        let okCount = 0, cancelCount = 0, failCount = 0;
        for (const file of picked) {
            let name = String(file.name || '').split(/[\\/]/).pop().replace(/['"`$\\<>]/g, '').trim();
            if (!name || name === '.' || name === '..') { results.push('跳过无效文件名: ' + (file.name || '(未知)')); failCount++; continue; }
            let targetPath;
            if (name === 'cns配置.ini' || name === '生成配置.sh') {
                targetPath = CNS_MODE_DIR + '/' + name;
                await runShellWithRoot(`mkdir -p ${shQuote(CNS_MODE_DIR)}`);
            } else {
                targetPath = CNS_DIR + '/' + name;
            }
            try {
                const existsRes = await runShellWithRoot(`[ -f ${shQuote(targetPath)} ] && echo EXISTS || echo NO`);
                if (parseResult(existsRes).includes('EXISTS') && !(await showCnsConfirm('已存在 ' + name + '，是否覆盖？', '覆 盖', true))) {
                    results.push('已取消: ' + name);
                    cancelCount++;
                    continue;
                }
                // 优先走 UFI 自带上传通道（不受 shell 单参数长度限制，二进制/大文件都稳）
                let handled = false;
                if (typeof uploadFileKano === 'function') {
                    try {
                        const upName = await uploadFileKano(file, false);
                        if (upName) {
                            const src = `/data/data/com.minikano.f50_sms/files/uploads/${upName}`;
                            const cpOut = parseResult(await runShellWithRoot(
                                `cp -f ${shQuote(src)} ${shQuote(targetPath)} && [ -f ${shQuote(targetPath)} ] && echo __OK__`, 120000
                            ));
                            if (cpOut.includes('__OK__')) handled = true;
                        }
                    } catch(e) { /* 落到 base64 分块兜底 */ }
                }
                if (!handled) {
                    if (file.size > 8 * 1024 * 1024) {
                        results.push('失败: ' + name + '（文件过大且上传通道不可用）');
                        failCount++;
                        continue;
                    }
                    const b64 = await readFileAsBase64(file);
                    const ret = await writeB64String(targetPath, b64);
                    if (!ret.ok) {
                        results.push('失败: ' + name + ' (' + (ret.out || '写入未完成') + ')');
                        failCount++;
                        continue;
                    }
                }
                results.push('成功: ' + name);
                okCount++;
            } catch(e) {
                results.push('失败: ' + name + ' (' + String(e) + ')');
                failCount++;
            }
        }
        appendLog('上传结束\n' + results.join('\n'));
        if (okCount > 0) showConfigToast('上传完成 ' + okCount + ' 个');
        else if (cancelCount > 0 && failCount === 0) showConfigToast('已取消上传');
        else showConfigToast('上传失败');
        const sel = document.getElementById('cns_config_select');
        if (sel) {
            const files = await fillConfigSelect();
            if (!files.length) return;
            const lastUploaded = picked[picked.length - 1] ? picked[picked.length - 1].name : '';
            if (lastUploaded) {
                const found = files.find(f => f.name === lastUploaded || f.name === 'makeCNSMode/' + lastUploaded);
                sel.value = found ? found.name : files[0].name;
            } else {
                sel.value = files[0].name;
            }
            await loadConfigFile(sel.value);
        }
    }
