    function parseResult(res) {
        if (res === null || res === undefined) return '(无输出)';
        if (typeof res === 'object' && res.content !== undefined) return String(res.content).trim();
        if (typeof res === 'string') return res.trim();
        try { return JSON.stringify(res); } catch(e) { return String(res); }
    }

    function nowTimeStr() {
        const now = new Date();
        return `${now.getFullYear()}/${String(now.getMonth()+1).padStart(2,'0')}/${String(now.getDate()).padStart(2,'0')} ${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}:${String(now.getSeconds()).padStart(2,'0')}`;
    }

    function appendLog(text) {
        const logBox = document.getElementById('cns_log');
        if (logBox) { logBox.textContent = `[${nowTimeStr()}]\n${text}`; logBox.scrollTop = logBox.scrollHeight; }
    }

    // --- 通用小工具 ---
    function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

    // shell 单引号安全转义，防止链接中的特殊字符被当成命令
    function shQuote(s) { return "'" + String(s).replace(/'/g, "'\\''") + "'"; }

    function isValidHttpUrl(u) { return /^https?:\/\/[^\s'"\\<>]+$/i.test(u); }

    function fmtSize(bytes) {
        const n = Number(bytes) || 0;
        if (n < 1024) return n + ' B';
        if (n < 1048576) return (n / 1024).toFixed(1) + ' KB';
        return (n / 1048576).toFixed(2) + ' MB';
    }

    // 多步骤日志：add 追加一行，progress 原地刷新最后一行
    function makeStepLogger() {
        const lines = [];
        let progressIdx = -1;
        const render = () => appendLog(lines.join('\n'));
        return {
            add(text) { progressIdx = -1; lines.push(text); render(); },
            progress(text) {
                if (progressIdx < 0) { lines.push(text); progressIdx = lines.length - 1; }
                else lines[progressIdx] = text;
                render();
            }
        };
    }

        if (mainDot) {

    }

    async function isCnsRunning() {
        const scriptChk = await runShellWithRoot(`[ -f '${CNS_DIR}/start.sh' ] && echo EXISTS || echo NO`);
        const noScript = !parseResult(scriptChk).includes('EXISTS');

        let res = await runShellWithRoot(`pgrep -x '${CNS_PROCESS_NAME}' 2>/dev/null`);
        let out = parseResult(res).trim();
        if (/\d/.test(out)) {
            return { running: true, info: 'PID: ' + out.split(/\s+/).join(', '), noScript };
        }

        res = await runShellWithRoot(`pgrep -f '/data/cns' 2>/dev/null`);
        out = parseResult(res).trim();
        if (/\d/.test(out)) {
            return { running: true, info: 'PID: ' + out.split(/\s+/).join(', '), noScript };
        }

        res = await runShellWithRoot("ps 2>/dev/null | grep '/data/cns' | grep -v grep");
        out = parseResult(res).trim();
        if (out) {
            const pidMatch = out.match(/^\s*\S+\s+(\d+)/);
            const pid = pidMatch ? pidMatch[1] : '?';
            return { running: true, info: 'PID: ' + pid, noScript };
        }

        return { running: false, info: '', noScript };
    }

    function statusText(r) {
        if (r.noScript) return '未检测到 CNS 脚本，请先创建 /data/cns 并放入脚本';
        return r.running ? 'CNS 正在运行' + (r.info ? ' (' + r.info + ')' : '') : 'CNS 未运行';
    }

    async function runCnsScript(scriptName, actionDesc) {
        let path;
        if (scriptName === '生成配置.sh') {
            path = CNS_UP_SCRIPT;
        } else {
            path = CNS_DIR + '/' + scriptName;
        }
        const check = await runShellWithRoot(`[ -f '${path}' ] && echo EXISTS || echo NO`);
        if (!parseResult(check).includes('EXISTS')) {
            appendLog(`脚本 ${path} 不存在，无法${actionDesc || '执行'}`);
            return;
        }
        appendLog(`正在${actionDesc || '执行'} ${path} ...`);
        const res = await runShellWithRoot(`sh ${path}`, 30000);
        const out = parseResult(res);
        appendLog(out);
        if (scriptName === 'status.sh') {
            const running = /running|run|active|启动|运行/i.test(out);
            if (running !== undefined) updateStatus(running);
        }
        setTimeout(async () => {
            const st = await isCnsRunning();
            updateStatus(st.running);
        }, 1500);
    }

    // --- 检测公网IP ---
    async function checkPublicIP() {
        appendLog('正在检测公网IP (ip.3322.net) ...');
        // 优先使用 curl，若失败则使用 wget
        let cmd = `curl -s --connect-timeout 10 ip.3322.net 2>/dev/null`;
        let res = await runShellWithRoot(cmd, 15000);
        let ip = parseResult(res).trim();
        if (!ip || ip === '' || /error|fail|not found/i.test(ip)) {
            cmd = `wget -q -O - --timeout=10 ip.3322.net 2>/dev/null`;
            res = await runShellWithRoot(cmd, 15000);
            ip = parseResult(res).trim();
        }
        if (ip && !/error|fail|not found|timeout/i.test(ip) && /^\d+\.\d+\.\d+\.\d+$/.test(ip)) {
            appendLog('公网IP: ' + ip);
        } else {
            appendLog('获取IP失败，可能无网络连接或服务不可用');
        }
    }

    // --- 下载并安装 CNS ---
    async function downloadAndInstallCns(url) {
        const log = makeStepLogger();
        const btn = document.getElementById('btn_download');
        if (btn) btn.disabled = true;
        const fail = (msg, toast) => {
            log.add(msg + '\n(临时文件保留在 ' + DL_TMP_DIR + '，可重试或手动清理)');
            showConfigToast(toast || '下载失败');
        };
        try {
            log.add('下载地址: ' + url);
            await runShellWithRoot(`rm -rf ${shQuote(DL_TMP_DIR)}; mkdir -p ${shQuote(DL_TMP_DIR)}`);

            // 后台下载脚本：避免长时间下载被前端超时打断
            const dlScript = [
                '#!/bin/sh',
                'URL=' + shQuote(url),
                'OUT=' + shQuote(DL_ZIP),
                'LOG=' + shQuote(DL_LOG),
                'CODE=' + shQuote(DL_CODE),
                'if command -v curl >/dev/null 2>&1; then',
                '  curl -fsS -L --connect-timeout 15 -o "$OUT" "$URL" 2>"$LOG"',
                '  echo $? > "$CODE"',
                'elif command -v wget >/dev/null 2>&1; then',
                '  wget -T 20 -O "$OUT" "$URL" 2>"$LOG"',
                '  echo $? > "$CODE"',
                'else',
                '  echo "curl / wget 均不可用" > "$LOG"',
                '  echo 127 > "$CODE"',
                'fi',
                ''
            ].join('\n');
            await runShellWithRoot(`echo ${shQuote(utf8ToBase64(dlScript))} | base64 -d > ${shQuote(DL_SCRIPT)}`);

            const started = parseResult(await runShellWithRoot(
                `if command -v nohup >/dev/null 2>&1; then (nohup sh ${shQuote(DL_SCRIPT)} >/dev/null 2>&1 &); else (sh ${shQuote(DL_SCRIPT)} >/dev/null 2>&1 &); fi; echo STARTED`
            ));
            if (!started.includes('STARTED')) return fail('× 无法启动下载任务');

            let waited = 0, code = null;
            while (waited < DL_MAX_WAIT_MS) {
                await sleep(1500);
                waited += 1500;
                const st = parseResult(await runShellWithRoot(
                    `if [ -f ${shQuote(DL_CODE)} ]; then printf 'DONE:'; cat ${shQuote(DL_CODE)}; elif [ -f ${shQuote(DL_ZIP)} ]; then printf 'SIZE:'; wc -c < ${shQuote(DL_ZIP)}; else printf 'SIZE:0'; fi`
                )).trim();
                if (st.indexOf('DONE:') === 0) {
                    code = parseInt(st.replace(/[^0-9-]/g, ''), 10);
                    break;
                }
                const size = parseInt(st.replace(/[^0-9]/g, ''), 10) || 0;
                log.progress('下载中... 已接收 ' + fmtSize(size) + '  (' + Math.round(waited / 1000) + 's)');
            }

            if (code === null) {
                await runShellWithRoot(`pkill -f ${shQuote(DL_SCRIPT)} 2>/dev/null; pkill -f ${shQuote(DL_ZIP)} 2>/dev/null`);
                return fail('× 下载超时（超过 ' + Math.round(DL_MAX_WAIT_MS / 1000) + ' 秒），已中止', '下载超时');
            }
            if (code !== 0) {
                const errLog = parseResult(await runShellWithRoot(`head -c 500 ${shQuote(DL_LOG)} 2>/dev/null`));
                return fail('× 下载失败 (退出码 ' + code + ')' + (errLog && errLog !== '(无输出)' ? '\n' + errLog : ''));
            }

            const sizeOut = parseResult(await runShellWithRoot(
                `if [ -f ${shQuote(DL_ZIP)} ]; then wc -c < ${shQuote(DL_ZIP)}; else echo 0; fi`
            )).replace(/[^0-9]/g, '');
            const zipSize = parseInt(sizeOut, 10) || 0;
            if (zipSize <= 0) return fail('× 下载文件为空，请检查链接');
            log.add('✓ 下载完成，已重命名为 cns.zip (' + fmtSize(zipSize) + ')');

            // 校验是否为 zip（PK 头），避免把网页/重定向页当成压缩包
            const magic = parseResult(await runShellWithRoot(`head -c 2 ${shQuote(DL_ZIP)} 2>/dev/null`));
            if (magic.indexOf('PK') !== 0) {
                const preview = parseResult(await runShellWithRoot(`head -c 200 ${shQuote(DL_ZIP)} 2>/dev/null`));
                return fail('× 下载内容不是 zip 压缩包，请确认是直链下载地址\n内容预览: ' + preview, '不是 zip 文件');
            }

            log.add('正在解压...');
            const tool = parseResult(await runShellWithRoot(
                `if command -v unzip >/dev/null 2>&1; then echo UNZIP; elif command -v busybox >/dev/null 2>&1; then echo BUSYBOX; else echo NONE; fi`
            ));
            if (tool.includes('NONE')) return fail('× 设备上没有 unzip（也没有 busybox），无法解压', '缺少 unzip');
            const unzipBin = tool.includes('UNZIP') ? 'unzip' : 'busybox unzip';
            await runShellWithRoot(`mkdir -p ${shQuote(DL_EXTRACT)}`);
            const unzipOut = parseResult(await runShellWithRoot(
                `${unzipBin} -o ${shQuote(DL_ZIP)} -d ${shQuote(DL_EXTRACT)} 2>&1`, 120000
            ));
            const entries = parseResult(await runShellWithRoot(`ls -A ${shQuote(DL_EXTRACT)} 2>/dev/null`))
                .split('\n').map(s => s.trim())
                .filter(s => s && s !== '__MACOSX' && s !== '.DS_Store' && s !== '(无输出)');
            if (!entries.length) return fail('× 解压失败\n' + unzipOut, '解压失败');

            // 压缩包内只有一个顶层目录时，取该目录作为 cns 目录内容
            let payload = DL_EXTRACT;
            if (entries.length === 1) {
                const only = DL_EXTRACT + '/' + entries[0];
                const isDir = parseResult(await runShellWithRoot(`[ -d ${shQuote(only)} ] && echo DIR || echo FILE`));
                if (isDir.includes('DIR')) payload = only;
            }
            log.add('✓ 解压完成' + (payload === DL_EXTRACT ? '' : '（顶层目录: ' + entries[0] + '）'));

            const dirExists = parseResult(await runShellWithRoot(`[ -d ${shQuote(CNS_DIR)} ] && echo EXISTS || echo NO`)).includes('EXISTS');
            if (dirExists) {
                if (!(await showCnsConfirm(CNS_DIR + ' 已存在。\n继续将停止运行中的 CNS 并删除原目录后覆盖安装（原配置会丢失），确定吗？', '覆盖安装', true))) {
                    log.add('已取消安装，下载文件保留在 ' + DL_ZIP);
                    showConfigToast('已取消安装');
                    return;
                }
                log.add('正在停止旧的 CNS 并删除原目录...');
                await runShellWithRoot(`if [ -f '${CNS_DIR}/stop.sh' ]; then sh '${CNS_DIR}/stop.sh'; fi; pkill -x '${CNS_PROCESS_NAME}' 2>/dev/null; pkill -f '${CNS_DIR}' 2>/dev/null`);
                await runShellWithRoot(`rm -rf ${shQuote(CNS_DIR)}`);
            }

            const mvOut = parseResult(await runShellWithRoot(`mv ${shQuote(payload)} ${shQuote(CNS_DIR)} 2>&1`));
            const installed = parseResult(await runShellWithRoot(`[ -d ${shQuote(CNS_DIR)} ] && echo EXISTS || echo NO`)).includes('EXISTS');
            if (!installed) return fail('× 移动到 ' + CNS_DIR + ' 失败' + (mvOut && mvOut !== '(无输出)' ? '\n' + mvOut : ''), '安装失败');
            log.add('✓ 文件夹已放入 /data/ 并命名为 cns');

            const chmodOut = parseResult(await runShellWithRoot(`chmod -R 777 ${shQuote(CNS_DIR)} 2>&1`));
            if (chmodOut && chmodOut !== '(无输出)' && /denied|permission|not permitted|error|失败/i.test(chmodOut)) {
                log.add('△ 赋权提示: ' + chmodOut);
            } else {
                log.add('✓ 已对 ' + CNS_DIR + ' 内所有文件赋予 777 权限');
            }

            const fileList = parseResult(await runShellWithRoot(`ls -A ${shQuote(CNS_DIR)} 2>/dev/null | head -n 40`));
            log.add('目录内容:\n' + fileList);

            await runShellWithRoot(`rm -rf ${shQuote(DL_TMP_DIR)}`);
            try { localStorage.setItem(DL_URL_KEY, url); } catch(e) {}
            showConfigToast('CNS 安装完成');

            const infoEl = document.getElementById('cns_info');
            if (infoEl) infoEl.textContent = ' 已安装';
            const st = await isCnsRunning();
            updateStatus(st.running);
            updateAutostartBtn(await isAutostartEnabled());
        } catch(e) {
            fail('× 下载安装出错: ' + String(e));
        } finally {
            if (btn) btn.disabled = false;
        }
    }

    function utf8ToBase64(str) {
        if (window.TextEncoder) {
            const bytes = new TextEncoder().encode(str);
            let bin = '';
            for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
            return btoa(bin);
        }
        return btoa(unescape(encodeURIComponent(str)));
    }
    function arrayBufferToBase64(buffer) {
        const bytes = new Uint8Array(buffer);
        let bin = '';
        for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
        return btoa(bin);
    }
    function readFileAsBase64(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.onload = () => {
                try { resolve(arrayBufferToBase64(reader.result)); } catch(e) { reject(e); }
            };
            reader.onerror = () => reject(reader.error || new Error('读取文件失败'));
            reader.readAsArrayBuffer(file);
        });
    }

    async function fillConfigSelect() {
        const sel = document.getElementById('cns_config_select');
        const files = [];
        const res1 = await runShellWithRoot(`ls ${CNS_DIR} 2>/dev/null | grep -E '\\.(conf|ini|txt|md)$' || echo ''`);
        const rootFiles = parseResult(res1).split('\n').map(s => s.trim()).filter(Boolean);
        for (const f of rootFiles) {
            if (!files.some(item => item.name === f)) files.push({ name: f, path: CNS_DIR + '/' + f });
        }
        const res2 = await runShellWithRoot(`ls ${CNS_MODE_DIR} 2>/dev/null | grep -E '\\.(conf|ini|txt|md)$' || echo ''`);
        const modeFiles = parseResult(res2).split('\n').map(s => s.trim()).filter(Boolean);
        for (const f of modeFiles) {
            const fullName = 'makeCNSMode/' + f;
            if (!files.some(item => item.name === fullName)) {
                files.push({ name: fullName, path: CNS_MODE_DIR + '/' + f });
            }
        }
        const existsMode = await runShellWithRoot(`[ -f '${CNS_MODE_FILE}' ] && echo EXISTS || echo NO`);
        if (parseResult(existsMode).includes('EXISTS')) {
            if (!files.some(item => item.name === 'makeCNSMode/cns配置.ini')) {
                files.push({ name: 'makeCNSMode/cns配置.ini', path: CNS_MODE_FILE });
            }
        }
        files.sort((a, b) => {
            if (a.name === 'config.ini') return -1;
            if (b.name === 'config.ini') return 1;
            if (a.name === 'makeCNSMode/cns配置.ini') return -1;
            if (b.name === 'makeCNSMode/cns配置.ini') return 1;
            return a.name.localeCompare(b.name);
        });
        if (sel) {
            if (files.length) {
                sel.innerHTML = files.map(f => '<option value="' + f.name + '">' + f.name + '</option>').join('');
            } else {
                sel.innerHTML = '<option value="">无配置文件</option>';
            }
        }
        return files;
    }

    async function loadConfigFile(fileName) {

    async function uploadConfigFiles(picked) {
        if (!picked.length) return;
        appendLog('开始上传 ' + picked.length + ' 个文件...');
        const results = [];
        let okCount = 0, cancelCount = 0, failCount = 0;
        for (const file of picked) {
            let name = String(file.name || '').split(/[\\/]/).pop().replace(/['"`$\\]/g, '').trim();
            if (!name || name === '.' || name === '..') { results.push('跳过无效文件名: ' + (file.name || '(未知)')); failCount++; continue; }
            let targetPath;
            if (name === 'cns配置.ini' || name === '生成配置.sh') {
                targetPath = CNS_MODE_DIR + '/' + name;
                await runShellWithRoot(`mkdir -p '${CNS_MODE_DIR}'`);
            } else {
                targetPath = CNS_DIR + '/' + name;
            }
            try {
                const existsRes = await runShellWithRoot(`[ -f '${targetPath}' ] && echo EXISTS || echo NO`);
                if (parseResult(existsRes).includes('EXISTS') && !(await showCnsConfirm('已存在 ' + name + '，是否覆盖？', '覆 盖', true))) {
                    results.push('已取消: ' + name);
                    cancelCount++;
                    continue;
                }
                const b64 = await readFileAsBase64(file);
                const res = await runShellWithRoot(`echo '${b64}' | base64 -d > '${targetPath}'`);
                const out = parseResult(res).trim();
                if (out && /not found|no such|permission|denied|error|失败/i.test(out)) {
                    results.push('失败: ' + name + ' (' + out + ')');
                    failCount++;
                    continue;
                }
                results.push('成功: ' + name);
                okCount++;
            } catch(e) {
                results.push('失败: ' + name + ' (' + String(e) + ')');
                failCount++;
            }
        }
        appendLog('上传结束\n' + results.join('\n'));
        if (okCount > 0) showConfigToast('上传完成 ' + okCount + ' 个');
        else if (cancelCount > 0 && failCount === 0) showConfigToast('已取消上传');
        else showConfigToast('上传失败');
        const sel = document.getElementById('cns_config_select');
        if (sel) {
            const files = await fillConfigSelect();
            if (!files.length) return;
            const lastUploaded = picked[picked.length-1]?.name;
            if (lastUploaded) {
                const found = files.find(f => f.name === lastUploaded || f.name === 'makeCNSMode/'+lastUploaded);
                if (found) sel.value = found.name;
                else sel.value = files[0].name;
            } else {
                sel.value = files[0].name;
            }
            await loadConfigFile(sel.value);
        }
    }

                    <button id="btn_download" class="cns-btn">⬇下载 CNS</button>

        document.body.appendChild(modal);

        (async () => {
            appendLog('正在自动检测状态...');
            const st = await isCnsRunning();
            appendLog('状态: ' + statusText(st));
            updateStatus(st.running);
            updateAutostartBtn(await isAutostartEnabled());
            const infoEl = document.getElementById('cns_info');
            if (infoEl) {
                const dirCheck = await runShellWithRoot(`[ -d '${CNS_DIR}' ] && echo EXISTS || echo NO`);
                if (parseResult(dirCheck).includes('EXISTS')) {
                    infoEl.textContent = ' 已安装';
                } else {
                    infoEl.textContent = ' 未安装';
                }
            }
        })();

        document.getElementById('btn_start').onclick = async () => {
            const dirCheck = await runShellWithRoot(`[ -d '${CNS_DIR}' ] && echo EXISTS || echo NO`);
            if (!parseResult(dirCheck).includes('EXISTS')) {
                appendLog('目录 /data/cns 不存在，请先创建并放入脚本');
                return;
            }
            await runCnsScript('start.sh', '启动');
            setTimeout(async () => {
                const st = await isCnsRunning();
                updateStatus(st.running);
            }, 1000);
        };
        document.getElementById('btn_stop').onclick = async () => {
            await runCnsScript('stop.sh', '停止');
            setTimeout(async () => {
                const st = await isCnsRunning();
                updateStatus(st.running);
            }, 1000);
        };
        document.getElementById('btn_status').onclick = async () => {
            await runCnsScript('status.sh', '检测状态');
        };
        document.getElementById('btn_gen').onclick = async () => {
            const dirCheck = await runShellWithRoot(`[ -d '${CNS_MODE_DIR}' ] && echo EXISTS || echo NO`);
            if (!parseResult(dirCheck).includes('EXISTS')) {
                appendLog('目录 ' + CNS_MODE_DIR + ' 不存在，请先创建');
                return;
            }
            await runCnsScript('生成配置.sh', '生成配置');
            setTimeout(async () => {
                const sel = document.getElementById('cns_config_select');
                if (sel) {
                    const files = await fillConfigSelect();
                    if (files.length) {
                        const target = files.find(f => f.name === 'makeCNSMode/cns配置.ini');
                        if (target) sel.value = target.name;
                        else sel.value = files[0].name;
                        await loadConfigFile(sel.value);
                    }
                }
            }, 1000);
        };
        // 新增IP检测按钮
        document.getElementById('btn_ip').onclick = async () => {
            await checkPublicIP();
        };
        document.getElementById('btn_config').onclick = () => openConfigModal();
        document.getElementById('btn_upload').onclick = async () => {
            const dirCheck = await runShellWithRoot(`[ -d '${CNS_DIR}' ] && echo EXISTS || echo NO`);
            if (!parseResult(dirCheck).includes('EXISTS')) {
                appendLog('目录 /data/cns 不存在，请先创建');
                return;
            }
            pickConfigFiles();
        };
        // 新增下载 CNS 按钮
        document.getElementById('btn_download').onclick = () => showDownloadModal();
        document.getElementById('btn_autostart').onclick = async () => {
            const on = await isAutostartEnabled();
            await setAutostart(!on);
        };
        document.getElementById('btn_uninstall').onclick = () => uninstallCns();
        // 新增使用说明按钮
        document.getElementById('btn_help').onclick = () => showHelpModal();
        document.getElementById('btn_modal_close').onclick = () => modal.remove();
        modal.onclick = (e) => { if (e.target === modal) modal.remove(); };
    }

    const menuBox = collapseBtn_menu?.nextElementSibling?.querySelector('.collapse_box');
    if (menuBox) {
        const entryBtn = document.createElement('button');
        entryBtn.innerHTML = 'CNS脚本控制<span class="cns-main-dot" id="main_status_dot"></span>';
        entryBtn.onclick = showMainModal;
        menuBox.appendChild(entryBtn);
    }

    (async () => {
        try {
            const st = await isCnsRunning();
            updateStatus(st.running);
        } catch(e) {}
    })();
})();
